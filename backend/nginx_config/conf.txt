server
{
    listen 9092;
    server_name _;
    index index.html index.htm default.htm default.html;
    root /home/www/mainWeb/dist;

    # 接口代理：转发到本机回环地址 127.0.0.1
    location ^~ /api {
        # 【核心修改】改为 127.0.0.1
        proxy_pass http://127.0.0.1:20000;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_http_version 1.1;
        proxy_set_header Connection "";
        limit_except GET POST PUT DELETE OPTIONS { deny all; }
    }

    # 二维码短码重定向代理
    location ^~ /qr/ {
        proxy_pass http://127.0.0.1:20000;
        
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }

    location / {
        if ($request_method = 'OPTIONS') {
            add_header Access-Control-Allow-Origin "http://cheerout.cn:9092";
            add_header Access-Control-Allow-Methods "GET,POST,PUT,DELETE,OPTIONS";
            add_header Access-Control-Allow-Headers "Content-Type,Authorization";
            add_header Access-Control-Allow-Credentials "true";
            add_header Content-Length 0;
            add_header Content-Type text/plain;
            return 200;
        }
        try_files $uri $uri/ /index.html;
    }

    # 【修正】Linux 正则去掉多余反斜杠
    location ~ .*\.(gif|jpg|jpeg|png|bmp|swf)$
    {
        expires      30d;
        access_log  /dev/null;
		error_log  /dev/null;
    }

    location ~ .*\.(js|css)?$
    {
        expires      12h;
        access_log  /dev/null;
		error_log  /dev/null;
    }

    location ~ ^/(\.user.ini|\.htaccess|\.git|\.env|\.svn|\.project|LICENSE|README.md)
    {
        return 404;
    }

    location ~ \.well-known{
        allow all;
    }
    if ( $uri ~ "^/\.well-known/.*\.(php|jsp|py|js|css|lua|ts|go|zip|tar\.gz|rar|7z|sql|bak)$" ) {
        return 403;
    }

    access_log  /www/wwwlogs/cheerout.cn.log;
    error_log  /www/wwwlogs/cheerout.cn.error.log;
}
